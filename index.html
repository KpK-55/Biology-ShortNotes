function doPost(e) {
  var file = getOrCreateFile_();
  file.setContent(e.postData.contents);
  return ContentService.createTextOutput(JSON.stringify({ok:true, updated:new Date().toISOString()})).setMimeType(ContentService.MimeType.JSON);
}
function doGet(e) {
  var file = getOrCreateFile_();
  return ContentService.createTextOutput(file.getBlob().getDataAsString() || "{}").setMimeType(ContentService.MimeType.JSON);
}
function getOrCreateFile_() {
  var name = "NEET_BioNotes_Cloud_Backup.json";
  var files = DriveApp.getFilesByName(name);
  return files.hasNext() ? files.next() : DriveApp.createFile(name, "{}", MimeType.PLAIN_TEXT);
}
